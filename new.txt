def analyze_all_tickets(tickets):

    logger.info("Analyzing Tickets...")

    results = []

    with ThreadPoolExecutor(

        max_workers=MAX_WORKERS

    ) as executor:

        futures = [

            executor.submit(

                analyze_ticket,

                ticket

            )

            for ticket in tickets

        ]

        for future in as_completed(futures):

            results.append(

                future.result()

            )

    logger.info(

        f"Completed {len(results)} Tickets"

    )

    return pd.DataFrame(results)

def generate_dashboard(df):

    dashboard = {}

    # OVERALL STATS
    #######################################################################

    dashboard["Total Tickets"] = len(df)

    dashboard["Total Users"] = df["Requested For"].nunique()

    dashboard["Total Regions"] = df["Client Region"].nunique()

    dashboard["Total Champions"] = df["Champion"].nunique()

    dashboard["Total Departments"] = df["Department"].nunique()

    dashboard["Total Assignment Groups"] = df["Assignment Group"].nunique()

    #######################################################################
    # ISSUE SUMMARY
    #######################################################################

    issue_summary = (
        df.groupby("Category")
        .size()
        .reset_index(name="Count")
        .sort_values("Count", ascending=False)
    )

        issue_summary["Count"]
        / len(df)

def export_excel(ticket_df, dashboard):

    logger.info("Exporting Excel...")

    with pd.ExcelWriter(
        OUTPUT_FILE,
        engine="xlsxwriter"
    ) as writer:

        workbook = writer.book

        ###################################################################
        # FORMATS
        ###################################################################

        header = workbook.add_format({

            "bold": True,
            "bg_color": "#1F4E78",
            "font_color": "white",
            "border": 1,
            "align": "center",
            "valign": "vcenter"

        })

        cell = workbook.add_format({

            "text_wrap": True,
            "border": 1,
            "valign": "top"

        })

        title = workbook.add_format({

            "bold": True,
            "font_size": 16

        })

        ###################################################################
        # SHEET 1
        ###################################################################

        ticket_df.to_excel(

            writer,

            sheet_name="Ticket Analysis",

            index=False

        )

        ws = writer.sheets["Ticket Analysis"]

        for col_num, value in enumerate(ticket_df.columns):

            ws.write(

                0,

                col_num,

                value,

                header

            )

        for i, col in enumerate(ticket_df.columns):

            width = max(

                ticket_df[col]

                .astype(str)

                .map(len)

                .max(),

                len(col)

            )

            ws.set_column(

                i,

                i,

                min(width + 5, 50),

                cell

            )

        ws.freeze_panes(1, 0)
        ws.autofilter(
            0,
            0,
            len(ticket_df),
            len(ticket_df.columns) - 1
        )

        ###################################################################
        # DASHBOARD
        ###################################################################

        dashboard_ws = workbook.add_worksheet("Dashboard")

        writer.sheets["Dashboard"] = dashboard_ws

        row = 0

        dashboard_ws.write(

            title

        )

        row += 2

        dashboard_ws.write(row, 0, "Total Tickets")

        row += 1

        dashboard_ws.write(row, 0, "Total Users")
        dashboard_ws.write(row, 1, dashboard["Total Users"])

        row += 1

        dashboard_ws.write(row, 0, "Total Regions")
        dashboard_ws.write(row, 1, dashboard["Total Regions"])

        row += 1

        dashboard_ws.write(row, 0, "Total Champions")
        dashboard_ws.write(row, 1, dashboard["Total Champions"])

        row += 3

            writer,

            startrow=row,
