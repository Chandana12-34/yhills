def generate_high_level_categories(iteration_1_df):
    """
    Dynamically generate generic high-level categories
    from Iteration 1 results using the LLM.
    """

    records = []

    for _, row in iteration_1_df.iterrows():
        records.append({
            "Issue Summary": clean_text(row.get("B - Issue Summary", "")),
            "Category 1": clean_text(row.get("C - Category 1", "")),
            "Action Taken": clean_text(row.get("D - Action Taken", ""))
        })

    prompt = f"""
Analyze the following RITM data and generate a concise list of
generic high-level categories.

The categories must:
- Be derived dynamically from the provided data.
- Represent the broader business/technical nature of the requests.
- NOT simply copy Category 1 values.
- NOT contain RITM-specific names, IDs, people, dates, or other details.
- NOT be hardcoded or based on a predefined classification list.
- Group similar types of requests under meaningful generic categories.

RITM DATA:
{json.dumps(records, indent=2)}

Return ONLY valid JSON in exactly this format:

{{
    "high_level_categories": [
        "Category 1",
        "Category 2",
        "Category 3"
    ]
}}
"""

    response = call_llm(
        SYSTEM_PROMPT_ITERATION_2,
        prompt
    )

    result = extract_json(response)

    return result.get("high_level_categories", [])
