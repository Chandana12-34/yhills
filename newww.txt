# ============================================================
# CLEAN TEXT
# ============================================================

def clean_text(value):

    if value is None:
        return ""

    try:
        if pd.isna(value):
            return ""
    except Exception:
        pass

    value = str(value)

    # Remove null/control characters that can cause JSON problems.
    value = value.replace("\x00", "")

    # Normalize line endings.
    value = value.replace("\r\n", "\n")
    value = value.replace("\r", "\n")

    # Remove other invalid control characters while keeping
    # newline and tab.
    value = "".join(
        ch for ch in value
        if ch in ("\n", "\t") or ord(ch) >= 32
    )

    value = re.sub(
        r"[ \t]+",
        " ",
        value
    )

    value = re.sub(
        r"\n{3,}",
        "\n\n",
        value
    )

    return value.strip()


# ============================================================
# CLEAN LLM JSON RESPONSE
# ============================================================

def clean_json_response(text):

    if not text:
        return ""

    text = str(text).strip()

    # Remove ```json ... ```
    text = re.sub(
        r"^```json\s*",
        "",
        text,
        flags=re.IGNORECASE
    )

    # Remove ``` ... ```
    text = re.sub(
        r"^```\s*",
        "",
        text
    )

    text = re.sub(
        r"\s*```$",
        "",
        text
    )

    return text.strip()


# ============================================================
# EXTRACT JSON
# ============================================================

def extract_json(text):

    text = clean_json_response(text)

    # Direct JSON
    try:
        return json.loads(text, strict=False)

    except json.JSONDecodeError:
        pass

    # JSON object embedded inside additional text
    start = text.find("{")
    end = text.rfind("}")

    if start == -1 or end == -1 or end <= start:
        raise ValueError(
            "No JSON object found in LLM response."
        )

    json_text = text[
        start:end + 1
    ]

    try:
        return json.loads(
            json_text,
            strict=False
        )

    except json.JSONDecodeError as e:

        print()
        print("========== JSON PARSE ERROR ==========")
        print(e)
        print("RAW LLM RESPONSE:")
        print(text[:10000])
        print("======================================")
        print()

        raise

# ============================================================
# CALL LLM
# ============================================================

def call_llm(
    system_prompt,
    user_prompt
):

    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {API_KEY}"
    }

    payload = {
        "model": MODEL_NAME,
        "temperature": 0,
        "messages": [
            {
                "role": "system",
                "content": system_prompt
            },
            {
                "role": "user",
                "content": user_prompt
            }
        ]
    }

    last_error = None

    for attempt in range(
        1,
        MAX_RETRIES + 1
    ):

        try:

            response = requests.post(
                API_URL,
                headers=headers,
                json=payload,
                timeout=REQUEST_TIMEOUT,
                verify=False
            )

            # Print the real API response for 4xx/5xx errors.
            if not response.ok:

                print()
                print(
                    "========== LLM API ERROR =========="
                )

                print(
                    "HTTP STATUS:",
                    response.status_code
                )

                print(
                    "RESPONSE:",
                    response.text[:5000]
                )

                print(
                    "==================================="
                )
                print()

            response.raise_for_status()

            response_json = response.json()

            try:

                content = (
                    response_json
                    ["choices"][0]
                    ["message"]
                    ["content"]
                )

            except (
                KeyError,
                IndexError,
                TypeError
            ):

                raise ValueError(
                    "Unexpected LLM response:\n"
                    + json.dumps(
                        response_json,
                        indent=2
                    )
                )

            return content

        except Exception as e:

            last_error = e

            if attempt < MAX_RETRIES:

                wait_time = attempt * 3

                print(
                    f"LLM request failed. "
                    f"Retry {attempt}/{MAX_RETRIES - 1} "
                    f"in {wait_time}s..."
                )

                time.sleep(
                    wait_time
                )

            else:

                print(
                    f"LLM request failed after "
                    f"{MAX_RETRIES} attempts."
                )

    raise last_error

# ANALYZE ONE RITM

def analyze_iteration_2(row):

    ritm = clean_text(
        row["A - RITM"]
    )

    print(
        f"Iteration 2 -> Processing {ritm}"
    )

    try:

        prompt = build_iteration_2_prompt(
            row
        )

        response = call_llm(
            SYSTEM_PROMPT_ITERATION_2,
            prompt
        )

        result = extract_json(
            response
        )

        high_level_category = clean_text(
            result.get(
                "E - High Level Categorisation",
                ""
            )
        )

        high_level_action = clean_text(
            result.get(
                "F - High Level Action Taken",
                ""
            )
        )

        return {

            "A - RITM":
                ritm,

            "B - Issue Summary":
                clean_text(
                    row["B - Issue Summary"]
                ),

            "C - Category 1":
                clean_text(
                    row["C - Category 1"]
                ),

            "D - Action Taken":
                clean_text(
                    row["D - Action Taken"]
                ),

            "E - High Level Categorisation":
                high_level_category,

            "F - High Level Action Taken":
                high_level_action
        }

    except Exception as e:

        print(
            f"ERROR Iteration 2 -> {ritm}"
        )

        print(
            str(e)
        )

        # Preserve Iteration 1 data even when Iteration 2 fails.
        return {

            "A - RITM":
                ritm,

            "B - Issue Summary":
                clean_text(
                    row["B - Issue Summary"]
                ),

            "C - Category 1":
                clean_text(
                    row["C - Category 1"]
                ),

            "D - Action Taken":
                clean_text(
                    row["D - Action Taken"]
                ),

            "E - High Level Categorisation":
                "ERROR",

            "F - High Level Action Taken":
                "ERROR"
        }

# ============================================================

def save_final_excel(
    df,
    output_file
):

    with pd.ExcelWriter(
        output_file,
        engine="xlsxwriter"
    ) as writer:

        df.to_excel(
            writer,
            sheet_name="Final Analysis",
            index=False
        )

        workbook = writer.book

        worksheet = writer.sheets[
            "Final Analysis"
        ]

        header_format = workbook.add_format({
            "bold": True,
            "bg_color": "#CFE2F3",
            "border": 1,
            "text_wrap": True,
            "valign": "top"
        })

        cell_format = workbook.add_format({
            "text_wrap": True,
            "valign": "top",
            "border": 1
        })

        # Header formatting
        for col_num, column in enumerate(
            df.columns
        ):

            worksheet.write(
                0,
                col_num,
                column,
                header_format
            )

        # Cell formatting
        for row_num in range(
            1,
            len(df) + 1
        ):

            for col_num in range(
                len(df.columns)
            ):

                value = df.iloc[
                    row_num - 1,
                    col_num
                ]

                if pd.isna(value):
                    value = ""

                worksheet.write(
                    row_num,
                    col_num,
                    value,
                    cell_format
                )

        # Column widths
        widths = {
            "A:A": 20,
            "B:B": 50,
            "C:C": 35,
            "D:D": 60,
            "E:E": 40,
            "F:F": 60
        }

        for column_range, width in widths.items():

            worksheet.set_column(
                column_range,
                width,
                cell_format
            )

        worksheet.freeze_panes(
            1,
            0
        )

        worksheet.autofilter(
            0,
            0,
            len(df),
            len(df.columns) - 1
        )

# RESUME SUPPORT
# ============================================================

def load_existing_results():

    if not RESUME_ENABLED:
        return {}

    if not Path(
        FINAL_OUTPUT_FILE
    ).exists():

        return {}

    try:

        existing_df = pd.read_excel(
            FINAL_OUTPUT_FILE,
            sheet_name="Final Analysis",
            engine="openpyxl"
        )

    except Exception as e:

        print(
            "Could not read existing final output."
        )

        print(
            str(e)
        )

        return {}

    required = set(
        FINAL_COLUMNS
    )

    if not required.issubset(
        existing_df.columns
    ):

        return {}

    results = {}

    for _, row in existing_df.iterrows():

        ritm = clean_text(
            row["A - RITM"]
        )

        category = clean_text(
            row[
                "E - High Level Categorisation"
            ]
        )

        action = clean_text(
            row[
                "F - High Level Action Taken"
            ]
        )

        # Only consider a row completed if both Iteration 2
        # fields contain a non-error value.
        if (
            ritm
            and category
            and action
            and category != "ERROR"
            and action != "ERROR"
        ):

            results[ritm] = {
                "A - RITM": ritm,
                "B - Issue Summary":
                    clean_text(
                        row["B - Issue Summary"]
                    ),
                "C - Category 1":
                    clean_text(
                        row["C - Category 1"]
                    ),
                "D - Action Taken":
                    clean_text(
                        row["D - Action Taken"]
                    ),
                "E - High Level Categorisation":
                    category,
                "F - High Level Action Taken":
                    action
            }

    print(
        f"Resume: {len(results)} completed RITMs "
        f"found in existing output."
    )

    return results
