# ============================================================
# FINAL CLASSIFICATION HIERARCHY ENFORCEMENT
# ============================================================

issue_type_key = " ".join(issue_type.split()).lower()

category_key = " ".join(normalized_category.split()).lower()
subcategory_key = " ".join(normalized_subcategory.split()).lower()

# ------------------------------------------------------------
# Application Support
# Required:
# Issue Type  = Application Support
# Category    = Application Support
# Sub Category = Functional Support / Technical Support
# ------------------------------------------------------------
if issue_type_key == "application support":

    normalized_category = "Application Support"

    # Handle correct LLM output
    if subcategory_key == "functional support":
        normalized_subcategory = "Functional Support"

    elif subcategory_key == "technical support":
        normalized_subcategory = "Technical Support"

    # Handle reversed LLM output:
    # Functional Support -> Application Support
    elif (
        category_key == "functional support"
        and subcategory_key == "application support"
    ):
        normalized_category = "Application Support"
        normalized_subcategory = "Functional Support"

    # Handle reversed LLM output:
    # Technical Support -> Application Support
    elif (
        category_key == "technical support"
        and subcategory_key == "application support"
    ):
        normalized_category = "Application Support"
        normalized_subcategory = "Technical Support"

    else:
        normalized_subcategory = ""

# ------------------------------------------------------------
# Application Issue
# Required:
# Category = selected Application Issue category
# Sub Category = same selected category
# ------------------------------------------------------------
elif issue_type_key == "application issue":

    if normalized_category:
        normalized_subcategory = normalized_category
