# ============================================================
# FINAL TAXONOMY VALIDATION
# ============================================================

def enforce_taxonomy(row):
    issue_type = " ".join(
        str(row.get("Issue Type", "") or "").split()
    ).strip().lower()

    category = " ".join(
        str(row.get("Category", "") or "").split()
    ).strip()

    subcategory = " ".join(
        str(row.get("Sub Category", "") or "").split()
    ).strip()

    category_key = category.lower()
    subcategory_key = subcategory.lower()

    # Application Support
    if issue_type == "application support":

        if category_key == "functional support":
            category = "Application Support"
            subcategory = "Functional Support"

        elif category_key == "technical support":
            category = "Application Support"
            subcategory = "Technical Support"

        elif category_key == "application support":
            if subcategory_key == "functional support":
                subcategory = "Functional Support"
            elif subcategory_key == "technical support":
                subcategory = "Technical Support"

    # Application Issue
    elif issue_type == "application issue":

        if category:
            subcategory = category

    row["Category"] = category
    row["Sub Category"] = subcategory

    return row
