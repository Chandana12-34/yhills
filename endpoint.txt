import os
import io
import extract_msg

from docx import Document
from docx.shared import Inches

from bs4 import BeautifulSoup

from PIL import Image

try:
    import pytesseract
    OCR_ENABLED = True
except:
    OCR_ENABLED = False


def msg_to_docx(msg_file, output_docx):

    msg = extract_msg.Message(msg_file)

    doc = Document()

    # ===================================
    # EMAIL DETAILS
    # ===================================

    doc.add_heading("Email Details", level=1)

    doc.add_paragraph(f"From: {msg.sender}")
    doc.add_paragraph(f"To: {msg.to}")

    if msg.cc:
        doc.add_paragraph(f"CC: {msg.cc}")

    if msg.bcc:
        doc.add_paragraph(f"BCC: {msg.bcc}")

    doc.add_paragraph(f"Subject: {msg.subject}")
    doc.add_paragraph(f"Date: {msg.date}")

    # ===================================
    # EMAIL BODY
    # ===================================

    doc.add_heading("Email Content", level=1)

    html_body = None

    try:
        html_body = msg.htmlBody
    except:
        pass

    if html_body:

        soup = BeautifulSoup(html_body, "html.parser")

        text = soup.get_text("\n", strip=True)

        doc.add_paragraph(text)

    elif msg.body:

        doc.add_paragraph(msg.body)

    # ===================================
    # INLINE IMAGES + ATTACHMENTS
    # ===================================

    doc.add_page_break()
    doc.add_heading("Images & Attachments", level=1)

    image_count = 0

    for att in msg.attachments:

        try:

            filename = att.longFilename or att.shortFilename

            if not filename:
                filename = f"attachment_{image_count}"

            data = att.data

            ext = filename.lower()

            if ext.endswith((".png", ".jpg", ".jpeg", ".bmp", ".gif")):

                image_count += 1

                image_path = f"temp_{image_count}_{filename}"

                with open(image_path, "wb") as f:
                    f.write(data)

                doc.add_picture(image_path, width=Inches(5))

                # OCR extraction
                if OCR_ENABLED:
                    try:
                        img = Image.open(image_path)
                        text = pytesseract.image_to_string(img)

                        if text.strip():
                            doc.add_paragraph(
                                "OCR Extracted Text:"
                            )
                            doc.add_paragraph(text)

                    except Exception as e:
                        print(e)

            else:

                doc.add_paragraph(
                    f"Attachment Found: {filename}"
                )

        except Exception as e:
            print(e)

    doc.save(output_docx)

    print("Saved:", output_docx)
