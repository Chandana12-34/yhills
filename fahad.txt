def normalize_category(category, issue_summary=""):
    """
    Normalize AI-generated categories into approved parent categories
    while preserving the specific issue topic.
    """

    category = str(category).strip()
    issue_summary = str(issue_summary).strip()

    text = f"{category} {issue_summary}".lower()

    # Application-related issues
    application_keywords = [
        "application issue",
        "application error",
        "application failure",
        "application problem",
        "application performance",
        "application workflow",
        "system error",
        "system failure",
        "application not working"
    ]

    if any(keyword in text for keyword in application_keywords):
        return "Application Issue"

    # Functional support
    functional_keywords = [
        "workflow exception",
        "workflow issue",
        "workflow problem",
        "decision making",
        "decision-making",
        "business process",
        "functional issue",
        "functional problem",
        "process issue"
    ]

    if any(keyword in text for keyword in functional_keywords):
        return "Functional Support"

    # Technical support
    technical_keywords = [
        "performance issue",
        "performance problem",
        "slow",
        "timeout",
        "technical issue",
        "technical problem",
        "integration issue",
        "interface issue",
        "api issue",
        "connectivity issue"
    ]

    if any(keyword in text for keyword in technical_keywords):
        return "Technical Support"

    # Preserve already-approved categories
    approved_categories = {
        "functional support",
        "technical support",
        "application issue"
    }

    if category.lower() in approved_categories:
        return category.title()

    # Default
    return category
