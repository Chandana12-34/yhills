def normalize_text(value):
    """Normalize text for reliable keyword matching."""
    if value is None:
        return ""

    return str(value).strip().lower()


def classify_application_support(summary):
    """
    Classify Application Support tickets into
    Category + Sub Category.
    """

    text = normalize_text(summary)

    # -------------------------------------------------
    # Functional Support subcategories
    # -------------------------------------------------

    if any(keyword in text for keyword in [
        "performance",
        "slow",
        "slowness",
        "timeout",
        "takes too long",
        "response time"
    ]):
        return "Functional Support", "Performance and Slowness"

    if any(keyword in text for keyword in [
        "decision making",
        "decision stage",
        "decision",
        "decision rule"
    ]):
        return "Functional Support", \
               "Decision Making (Issue faced at Decision Stage)"

    if any(keyword in text for keyword in [
        "data flow",
        "internal flow",
        "data between tasks",
        "data between stages"
    ]):
        return "Functional Support", \
               "Data Flow Internal (Internal flow of data between tasks and stages)"

    if any(keyword in text for keyword in [
        "front office",
        "business power",
        "initiation",
        "validation"
    ]):
        return "Functional Support", \
               "Front Office Initiation/Validation (Business Power)"

    if any(keyword in text for keyword in [
        "hierarchy",
        "hierarchies"
    ]):
        return "Functional Support", \
               "Hierarchy (Issue at creation or display of Hierarchy)"

    if any(keyword in text for keyword in [
        "kyc",
        "escalation form"
    ]):
        return "Functional Support", \
               "KYC and Escalation Form"

    if any(keyword in text for keyword in [
        "screening",
        "screening issue"
    ]):
        return "Functional Support", "Screening Issue"

    if any(keyword in text for keyword in [
        "user interface",
        "ui issue",
        "ui",
        "wrong data in ui",
        "missing data in ui",
        "missing field",
        "field not displayed",
        "display issue"
    ]):
        return "Functional Support", \
               "User Interface (Issue with wrong or missing data in UI)"

    if any(keyword in text for keyword in [
        "workflow exception",
        "workflow issue",
        "workflow error",
        "workflow"
    ]):
        return "Functional Support", "Workflow Exception"

    # -------------------------------------------------
    # If it doesn't match a known functional category
    # -------------------------------------------------

    return "Functional Support", "Other"
