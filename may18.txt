import os
import win32com.client

# INPUT AND OUTPUT FOLDERS
INPUT_FOLDER = r"C:\Users\YourUserName\Documents\FENERGO (2)"
OUTPUT_FOLDER = r"C:\Users\YourUserName\Documents\FENERGO_Output"

os.makedirs(OUTPUT_FOLDER, exist_ok=True)

# Start Word
word = win32com.client.Dispatch("Word.Application")
word.Visible = False
word.DisplayAlerts = 0

processed = 0
failed = 0

for filename in os.listdir(INPUT_FOLDER):

    if not filename.lower().endswith(".docx"):
        continue

    source_file = os.path.join(INPUT_FOLDER, filename)
    output_file = os.path.join(OUTPUT_FOLDER, filename)

    try:
        print(f"Processing: {filename}")

        doc = word.Documents.Open(source_file)

        # Save into output folder
        doc.SaveAs2(output_file)

        try:
            label = doc.SensitivityLabel

            print("Current Label:", label)

            # Try to display label information
            try:
                print("Label Name:", label.Name)
            except:
                pass

        except Exception as e:
            print("Sensitivity label API unavailable:", e)

        doc.Close(False)

        processed += 1

    except Exception as e:
        failed += 1
        print(f"FAILED: {filename}")
        print(e)

word.Quit()

print("\n========================")
print(f"Processed : {processed}")
print(f"Failed    : {failed}")
print("========================")
