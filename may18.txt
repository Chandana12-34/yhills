markdown = markdown.replace("**", "")
markdown = markdown.replace("### ", "")
markdown = markdown.replace("## ", "")
markdown = markdown.replace("# ", "")

lines = markdown.split("\n")

table_lines = []

for line in lines:

    line = line.strip()

    if not line:
        continue

    # TABLE DETECTION
    if "|" in line:

        table_lines.append(line)
        continue

    # CREATE TABLE
    if table_lines:

        rows = []

        for tbl_line in table_lines:

            if "---" in tbl_line:
                continue

            cols = [
                x.strip()
                for x in tbl_line.split("|")
                if x.strip()
            ]

            if cols:
                rows.append(cols)

        if rows:

            table = doc.add_table(
                rows=len(rows),
                cols=len(rows[0])
            )

            table.style = "Table Grid"

            for r in range(len(rows)):
                for c in range(len(rows[r])):
                    table.cell(r, c).text = rows[r][c]

        table_lines = []

    # BULLETS
    if line.startswith("- "):

        doc.add_paragraph(
            line[2:],
            style="List Bullet"
        )

    # NUMBERED LIST
    elif line[:2].isdigit():

        doc.add_paragraph(
            line,
            style="List Number"
        )

    # NORMAL TEXT
    else:

        doc.add_paragraph(line)

# LAST TABLE
if table_lines:

    rows = []

    for tbl_line in table_lines:

        if "---" in tbl_line:
            continue

        cols = [
            x.strip()
            for x in tbl_line.split("|")
            if x.strip()
        ]

        if cols:
            rows.append(cols)

            cols=len(rows[0])
        )
