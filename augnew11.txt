issue_type_df = df.copy()

issue_type_df["Issue Type"] = (
    issue_type_df["Issue Type"]
    .fillna("Other")
    .astype(str)
    .str.strip()
    .replace("", "Other")
)

issue_type_summary = (
    issue_type_df
    .groupby("Issue Type", dropna=False)
    .size()
    .reset_index(name="Count")
    .sort_values("Count", ascending=False)
)

issue_type_summary["Percentage"] = (
    issue_type_summary["Count"] / len(issue_type_df) * 100
).round(2)

dashboard["issue_type_summary"] = issue_type_summary
