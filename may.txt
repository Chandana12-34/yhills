def get_application_status(app_name, env_name, cfg):
    """
    Returns True only if every endpoint in the selected application is UP.
    """

    env_cfg = cfg.applications[app_name].environments[env_name]

    for ep in env_cfg.endpoints:

        endpoint_data = st.session_state.health_data.get(ep.name)

        if endpoint_data is None:
            return False

        status = str(endpoint_data.get("status", "")).upper()

        if status != "UP":
            return False

    return True
