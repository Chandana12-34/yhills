def get_application_status(app_name, env_name):
    """
    Returns True only if every endpoint of the application is UP.
    """

    endpoints = st.session_state.health_data.get(app_name, {}).get(env_name, {})

    if not endpoints:
        return False

    for endpoint_name, endpoint_data in endpoints.items():

        status = str(endpoint_data.get("status", "")).upper()

        if status != "UP":
            return False

    return True
