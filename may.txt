##########################################################################
# REMOVE HTML
##########################################################################

def remove_html(text):

    if pd.isna(text):

        return ""

    text = str(text)

    text = re.sub("<.*?>", " ", text)

    return text

##########################################################################
# REMOVE EXTRA SPACES
##########################################################################

def clean_spaces(text):

    text = re.sub(r"\s+", " ", text)

    return text.strip()

##########################################################################
# REMOVE SERVICENOW METADATA
##########################################################################

def clean_servicenow_notes(text):

    if not text:

        return ""

    patterns = [

        r"Infos:.*?$",

        r"State\s*-\s*Open",

        r"State\s*-\s*On Hold",

        r"APS L1 Support",

        r"Updated by.*",

        r"Created by.*",

        r"Work notes.*",

        r"Additional comments.*"

    ]

    for p in patterns:

        text = re.sub(

            p,

            "",

            text,

            flags=re.IGNORECASE | re.MULTILINE

        )

    return clean_spaces(text)

##########################################################################
# HASH
##########################################################################

def generate_hash(short_desc, desc):

    txt = str(short_desc) + str(desc)

    return hashlib.sha256(

        txt.encode()

    ).hexdigest()

def find_column(df, aliases):

    cols = {c.lower().strip(): c for c in df.columns}

    for alias in aliases:
        if alias.lower() in cols:
            return cols[alias.lower()]

    return None

##########################################################################
# READ EXCEL
##########################################################################

def load_excel(file_path):

    logger.info("Reading Excel...")

    df = pd.read_excel(file_path)

    mapping = {}

    for key, aliases in COLUMN_ALIASES.items():

        mapping[key] = find_column(df, aliases)

        if mapping[key]:
            logger.info(f"{key} -> {mapping[key]}")
        else:
            logger.warning(f"{key} column not found")

    return df, mapping

##########################################################################
# PREPROCESS ROW
##########################################################################

def preprocess_ticket(row, mapping):

    number = row.get(mapping["number"], "")

    short_desc = remove_html(
        row.get(mapping["short_description"], "")
    )

    description = remove_html(
        row.get(mapping["description"], "")
    )

    comments = remove_html(
        row.get(mapping["comments"], "")
    )

    comments = clean_servicenow_notes(comments)

    assignment_group = row.get(
        mapping["assignment_group"],
        ""
    )

    requested_for = row.get(
        mapping["requested_for"],
        ""
    )

    department = row.get(
        mapping["department"],
        ""
    )

    region = row.get(
        mapping["client_region"],
        ""
    )

    entity = row.get(
        mapping["entity"],
        ""
    )

    champion = row.get(
        mapping["champion"],
        ""
    )

    priority = row.get(
        mapping["priority"],
        ""
    )

    created = row.get(
        mapping["created"],
        ""
    )

    updated = row.get(
        mapping["updated"],
        ""
    )

    combined_text = f"""

Short Description

{short_desc}

Description

{description}

Comments

{comments}

"""

    ticket = {

        "ritm": number,

        "short_description": short_desc,

        "description": description,

        "comments": comments,

        "combined_text": clean_spaces(combined_text),

        "assignment_group": assignment_group,

        "requested_for": requested_for,

        "department": department,

        "region": region,

        "entity": entity,

        "champion": champion,

        "priority": priority,

        "created": created,

        "updated": updated,

        "hash": generate_hash(
            short_desc,
            description
        )

    }

    return ticket

##########################################################################
# BUILD ALL TICKETS
##########################################################################

def prepare_tickets(df, mapping):

    logger.info("Preparing tickets...")

    tickets = []

    for _, row in df.iterrows():

        try:

            ticket = preprocess_ticket(row, mapping)

            tickets.append(ticket)

        except Exception as ex:

            logger.exception(ex)

    logger.info(f"Prepared {len(tickets)} tickets")

    return tickets
