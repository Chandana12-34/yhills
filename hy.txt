def normalize_category(category, issue_summary="", issue_type=""):
    """
    Normalize the parent Category.

    Final allowed Categories:
        - Application Issue
        - Functional Support
        - Technical Support

    The detailed issue classification belongs in Sub Category.
    """

    category = str(category or "").strip()
    issue_summary = str(issue_summary or "").strip().lower()
    issue_type = str(issue_type or "").strip().lower()

    # ============================================================
    # APPLICATION ISSUE
    # ============================================================

    if issue_type == "application issue":

        # If the incoming category is already an application
        # issue category, the parent category must still be
        # "Application Issue".
        return "Application Issue"

    # ============================================================
    # APPLICATION SUPPORT
    # ============================================================

    if issue_type == "application support":

        # Preserve an already normalized parent category
        if category.lower() in [
            "functional support",
            "technical support"
        ]:
            return category.title()

        # --------------------------------------------------------
        # FUNCTIONAL SUPPORT
        # --------------------------------------------------------

        if any(k in issue_summary for k in [
            "kyc",
            "escalation form",
            "functional",
            "business process",
            "business rule",
            "workflow",
            "functional issue",
            "functional problem",
            "decision making",
            "decision stage"
        ]):
            return "Functional Support"

        # --------------------------------------------------------
        # TECHNICAL SUPPORT
        # --------------------------------------------------------

        if any(k in issue_summary for k in [
            "technical",
            "integration",
            "interface",
            "api",
            "connectivity",
            "crds",
            "upstream",
            "downstream",
            "data synchronization",
            "data synchronisation",
            "synchronization",
            "synchronisation"
        ]):
            return "Technical Support"

        # Default
        return "Functional Support"

    # ============================================================
    # ALREADY NORMALIZED CATEGORY
    # ============================================================

    if category.lower() == "application issue":
        return "Application Issue"

    if category.lower() == "functional support":
        return "Functional Support"

    if category.lower() == "technical support":
        return "Technical Support"

    # ============================================================
    # FALLBACK
    # ============================================================

    return category if category else "Other"


def normalize_subcategory(category, subcategory="", issue_summary=""):
    """
    Normalize Sub Category according to the normalized parent Category.

    Parent Categories:
        Application Issue
        Functional Support
        Technical Support
    """

    category = str(category or "").strip()
    subcategory = str(subcategory or "").strip()
    issue_summary = str(issue_summary or "").strip().lower()

    category_lower = category.lower()
    subcategory_lower = subcategory.lower()

    # ============================================================
    # APPLICATION ISSUE SUB-CATEGORIES
    # ============================================================

    application_issue_subcategories = {
        "performance and slowness":
            "Performance and Slowness",

        "decision making (issue faced at decision stage)":
            "Decision Making (Issue faced at Decision Stage)",

        "data flow internal (internal flow of data between tasks and stages)":
            "Data Flow Internal (Internal flow of data between tasks and stages)",

        "front office initiation/validation (business power)":
            "Front Office Initiation/Validation (Business Power)",

        "hierarchy (issue at creation or display of hierarchy)":
            "Hierarchy (Issue at creation or display of Hierarchy)",

        "kyc and escalation form":
            "KYC and Escalation Form",

        "screening issue":
            "Screening Issue",

        "user interface (issue with wrong or missing data in ui)":
            "User Interface (Issue with wrong or missing data in UI)",

        "workflow exception":
            "Workflow Exception",

        "other":
            "Other",
    }

    if category_lower == "application issue":

        # --------------------------------------------------------
        # Preserve an existing approved sub-category
        # --------------------------------------------------------

        if subcategory_lower in application_issue_subcategories:
            return application_issue_subcategories[subcategory_lower]

        # --------------------------------------------------------
        # Infer from Issue Summary
        # --------------------------------------------------------

        if any(k in issue_summary for k in [
            "performance",
            "slow",
            "slowly",
            "slowness",
            "timeout",
            "taking too long",
            "takes too long"
        ]):
            return "Performance and Slowness"

        if any(k in issue_summary for k in [
            "decision making",
            "decision stage",
            "decision"
        ]):
            return "Decision Making (Issue faced at Decision Stage)"

        if any(k in issue_summary for k in [
            "data flow",
            "dataflow",
            "internal data flow",
            "data flow internal",
            "data not flowing",
            "data flow between",
            "between tasks",
            "between stages"
        ]):
            return "Data Flow Internal (Internal flow of data between tasks and stages)"

        if any(k in issue_summary for k in [
            "front office",
            "business power",
            "initiation",
            "validation"
        ]):
            return "Front Office Initiation/Validation (Business Power)"

        if any(k in issue_summary for k in [
            "hierarchy",
            "hierarchy creation",
            "hierarchy display"
        ]):
            return "Hierarchy (Issue at creation or display of Hierarchy)"

        if any(k in issue_summary for k in [
            "kyc",
            "escalation form"
        ]):
            return "KYC and Escalation Form"

        if any(k in issue_summary for k in [
            "screening",
            "screening issue"
        ]):
            return "Screening Issue"

        if any(k in issue_summary for k in [
            "user interface",
            "ui issue",
            "ui problem",
            "wrong data in ui",
            "missing data in ui",
            "display issue",
            "not visible",
            "missing on screen"
        ]):
            return "User Interface (Issue with wrong or missing data in UI)"

        if any(k in issue_summary for k in [
            "workflow exception",
            "workflow error",
            "workflow issue",
            "workflow problem",
            "workflow"
        ]):
            return "Workflow Exception"

        return "Other"

    # ============================================================
    # FUNCTIONAL SUPPORT
    # ============================================================

    if category_lower == "functional support":

        # --------------------------------------------------------
        # Preserve approved Functional Support sub-category
        # --------------------------------------------------------

        functional_subcategories = {
            "kyc and escalation form":
                "KYC and Escalation Form",

            "workflow exception":
                "Workflow Exception",

            "other":
                "Other",
        }

        if subcategory_lower in functional_subcategories:
            return functional_subcategories[subcategory_lower]

        # --------------------------------------------------------
        # Infer from Issue Summary / Description
        # --------------------------------------------------------

        if any(k in issue_summary for k in [
            "kyc",
            "escalation form",
            "escalation"
        ]):
            return "KYC and Escalation Form"

        if any(k in issue_summary for k in [
            "workflow exception",
            "workflow error",
            "workflow issue",
            "workflow problem",
            "workflow"
        ]):
            return "Workflow Exception"

        return "Other"

    # ============================================================
    # TECHNICAL SUPPORT
    # ============================================================

    if category_lower == "technical support":

        # --------------------------------------------------------
        # Approved Technical Support sub-category
        # --------------------------------------------------------

        if subcategory_lower == "up & downstream with crds":
            return "Up & Downstream with CRDS"

        # --------------------------------------------------------
        # Infer from Issue Summary / Description
        # --------------------------------------------------------

        if any(k in issue_summary for k in [
            "upstream",
            "downstream",
            "crds",
            "data synchronization",
            "data synchronisation",
            "synchronization",
            "synchronisation"
        ]):
            return "Up & Downstream with CRDS"

        return "Other"

    # ============================================================
    # FALLBACK
    # ============================================================

    return subcategory if subcategory else "Other"
