# Issue category breakdown
# ------------------------------------------------------------

# Make sure missing classification values are not dropped
summary_df = df.copy()

summary_df["Issue Type"] = (
    summary_df["Issue Type"]
    .fillna("Other")
    .astype(str)
    .str.strip()
    .replace("", "Other")
)

summary_df["Category"] = (
    summary_df["Category"]
    .fillna("Other")
    .astype(str)
    .str.strip()
    .replace("", "Other")
)

summary_df["Sub Category"] = (
    summary_df["Sub Category"]
    .fillna("Other")
    .astype(str)
    .str.strip()
    .replace("", "Other")
)

# Create summary WITHOUT dropping tickets
issue_summary = (
    summary_df
    .groupby(
        ["Issue Type", "Category", "Sub Category"],
        dropna=False
    )
    .size()
    .reset_index(name="Count")
    .sort_values(
        ["Issue Type", "Category", "Count", "Sub Category"],
        ascending=[True, True, False, True]
    )
)

# Percentage must be based on ALL processed tickets
issue_summary["Percentage"] = (
    issue_summary["Count"] / len(summary_df) * 100
).round(2)

dashboard["issue_summary"] = issue_summary
